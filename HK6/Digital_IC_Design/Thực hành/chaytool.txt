Check quyền server:
ps -ef | grep lmgrd
ps -ef | grep snpslmd

Chạy license:
/usr/synopsys/scl/2018.06/linux64/bin/lmgrd -c /usr/synopsys/Synopsys.dat
export SNPSLMD_LICENSE_FILE=27000@localhost

Chạy custom desgin (cnay chạy cuối cùng sau khi thêm hết license và path):
/usr/synopsys/customcompiler/N-2017.12-SP1-2/bin/custom_compiler


Copy thư viện từ project cũ
cp ../lib.defs .
cp ../tech.db .
cp ../display.tcl .

Cho DFF:
cp ./lab1vlsi/lib.defs .
cp ../lab1vlsi/tech.db .
cp ../lab1vlsi/display.tcl .


Thêm thư viện reference90RF:
echo "DEFINE reference90RF /root/lab1vlsi/PDK/reference90RF" >> lib.defs

export PATH=$PATH:/usr/synopsys/hspice/N-2017.12-SP2/hspice/bin

export PATH=$PATH:/usr/synopsys/wv/N-2017.12-SP2/platforms/linux64/bin

export LD_LIBRARY_PATH=$LD_LIBRARY_PATH:/usr/synopsys/wv/N-2017.12-SP2/platforms/linux64/lib
